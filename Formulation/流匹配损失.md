# 流匹配

# 为什么流匹配在语音编辑中越来越流行？
1. 少步推理更快：流匹配的 ODE 求解器（如 8~10 步 Euler/Heun）比 DDPM 的 50~100 步 DDIM 更快
2. 训练更稳定：连续时间 + 直推流路径，没有复杂的噪声调度设计
3. 更自然的条件注入：流匹配的速度场天然支持条件化，编辑任务的"参考引导"更容易融入


## 数学原理

**直推流（OT-Conditional Flow Matching）** ：

$$
x_t = (1-t)\cdot x_0 + t\cdot x_1 + \sigma(t)\cdot \epsilon
$$

其中 $x_0$ 是噪声样本，$x_1$ 是真实数据，$\sigma(t)$ 是噪声强度函数（通常取常数或线性衰减）。

对应的速度场（训练目标）为：

$$
u_t(x_t) = \frac{x_1 - x_0}{1} - \sigma'(t)\epsilon \bigg/ \frac{\mathrm{d}x_t}{\mathrm{d}t}
\quad \text{（简化形式：当 $\sigma(t)$ 为常数 $\sigma_\mathrm{min}$ 时）}
$$

更简洁的直推流形式（CosyEdit / CosyVoice 使用）：

$$
x_t = t\cdot x_1 + (1-t)\cdot x_0 + \sigma_\mathrm{min}\cdot z,\quad z \sim \mathcal{N}(0,\mathbf{I})
$$

$$
u_t = x_1 - x_0
$$

即速度就是目标和源的差，模型直接学习这个差值。

## 和扩散的区别

- **数学底层**：

扩散本质是 SDE 的逆向过程：

$$
dx = f(x,t)dt + g(t)dW
$$

正向加噪 → 逆向去噪（SDE 的逆仍是 SDE）

流匹配本质是 ODE 的向量场拟合：

$$
\frac{dx}{dt} = v_\theta(x_t, t)
$$

直接学习从 $p_0$ 到 $p_1$ 的确定性流动路径

import torch
import torch.nn as nn
import torch.nn.functional as F
import math
from typing import Optional, Tuple

class GotCFMWrapper(nn.Module):
    """
    GOT-CFM (Guided Optimal Transport - Conditional Flow Matching) 包装器
    包含：训练损失计算 + 推理采样
    """

    def __init__(
        self,
        model: nn.Module,       # GotCFM U-Net 主干
        sigma_min: float = 0.0,  # OT-CFM 的 σ_min
        sample_n_steps: int = 10,  # 推理采样步数
    ):
        super().__init__()
        self.model = model
        self.sigma_min = sigma_min
        self.sample_n_steps = sample_n_steps

    # ===== 1. 训练前向：计算流匹配损失 =====
    def forward(
        self,
        clean_mel_ori: torch.Tensor,   # 原始参考梅尔谱 [B, n_mels, T_ori]
        clean_mel_tar: torch.Tensor,   # 目标梅尔谱 [B, n_mels, T_tar]
        audio_tokens: torch.Tensor,     # 语义 token 索引 [B, L]
        spk_emb: torch.Tensor,          # 说话人嵌入 [B, D]
        mask_tar: Optional[torch.Tensor] = None,  # 目标掩码 [B, T_tar]
    ) -> dict:
        """
        CFM 训练过程：
        1. 随机采样时间 t ~ U(0, 1)
        2. 采样噪声 ε ~ N(0, I)
        3. 构造带噪样本: noisy_mel_tar = t * clean_mel + (1-t) * noise + σ_min * z
        4. 计算目标速度: target_vel = clean_mel - noise
        5. 模型预测速度 vel = model(noisy_mel, t, cond)
        6. 损失 = MSE(vel, target_vel)
        """
        B = clean_mel_tar.shape[0]
        device = clean_mel_tar.device

        # ---- 随机采样时间 t ~ U(0, 1) ----
        t = torch.rand(B, device=device, dtype=clean_mel_tar.dtype)

        # ---- 采样噪声 ----
        noise = torch.randn_like(clean_mel_tar)

        # ---- 构造带噪目标梅尔谱 (直推流) ----
        # noisy = t * x_1 + (1-t) * x_0 + σ_min * z
        # 其中 x_1 = clean_mel_tar (目标数据), x_0 = noise (源噪声)
        t_expand = t.view(B, 1, 1)  # [B, 1, 1] for broadcasting
        noisy_mel_tar = (
            t_expand * clean_mel_tar
            + (1 - t_expand) * noise
            + self.sigma_min * torch.randn_like(clean_mel_tar)
        )

        # ---- 目标速度 ----
        # u_t = x_1 - x_0 = clean_mel_tar - noise
        target_vel = clean_mel_tar - noise

        # ---- 模型预测速度 ----
        pred_vel, _ = self.model(
            noisy_mel_ori=clean_mel_ori,     # 原始参考（干净的，作为引导）
            noisy_mel_tar=noisy_mel_tar,     # 目标（带噪的）
            clean_mel_ori=clean_mel_ori,     # 干净原始（参考引导）
            masked_mel_tar=mask_tar,         # 目标掩码
            audio_tokens=audio_tokens,       # 语义 token
            t=t,                              # 时间步
            spk_emb=spk_emb,                  # 说话人嵌入
        )

        # ---- MSE 损失 ----
        if mask_tar is not None:
            # 只在有效区域计算损失
            mask_expand = mask_tar.unsqueeze(1).float()  # [B, 1, T]
            diff = (pred_vel - target_vel) ** 2
            loss = (diff * mask_expand).sum() / (mask_expand.sum() + 1e-8)
        else:
            loss = F.mse_loss(pred_vel, target_vel)

        return {"loss": loss, "pred_vel": pred_vel}

    # ===== 2. 推理采样：Euler 法求解 ODE =====
    @torch.no_grad()
    def sample(
        self,
        clean_mel_ori: torch.Tensor,  # 参考原始梅尔谱
        T_tar: int,                    # 目标时长（帧数）
        audio_tokens: torch.Tensor,    # 语义 token
        spk_emb: torch.Tensor,         # 说话人嵌入
        n_steps: Optional[int] = None,  # 采样步数
    ) -> Tuple[torch.Tensor, list]:
        """
        ODE 采样：从 t=0 的随机噪声出发，逐步积分到 t=1
        
        Euler 法: x_{t+Δt} = x_t + Δt * v_θ(x_t, t)
        
        Args:
            clean_mel_ori: 参考梅尔谱 [B, n_mels, T_ori]
            T_tar: 目标帧数
            audio_tokens: 语义 token [B, L]
            spk_emb: 说话人嵌入 [B, D]
            n_steps: 采样步数，默认用 self.sample_n_steps
        Returns:
            mel_tar: 生成的目标梅尔谱 [B, n_mels, T_tar]
            trajectory: 中间过程列表（用于可视化）
        """
        if n_steps is None:
            n_steps = self.sample_n_steps

        B = clean_mel_ori.shape[0]
        n_mels = clean_mel_ori.shape[1]
        device = clean_mel_ori.device
        dtype = clean_mel_ori.dtype

        # ---- 初始化: x_0 ~ N(0, I) ----
        mel_t = torch.randn(B, n_mels, T_tar, device=device, dtype=dtype)

        # ---- 时间步序列: 从 0 到 1，均匀步进 ----
        dt = 1.0 / n_steps
        time_steps = torch.linspace(0, 1, n_steps + 1, device=device, dtype=dtype)

        trajectory = [mel_t.clone()]

        # ---- Euler 积分 ----
        for i in range(n_steps):
            t = time_steps[i]
            t_batch = t.expand(B)  # [B]

            # 预测速度
            vel, _ = self.model(
                noisy_mel_ori=clean_mel_ori,
                noisy_mel_tar=mel_t,
                clean_mel_ori=clean_mel_ori,
                masked_mel_tar=None,
                audio_tokens=audio_tokens,
                t=t_batch,
                spk_emb=spk_emb,
            )

            # Euler 更新: x_{t+dt} = x_t + dt * v
            mel_t = mel_t + dt * vel

            trajectory.append(mel_t.clone())

        return mel_t, trajectory

    # ===== 3. 更高级的采样器：Heun 方法（二阶精度） =====
    @torch.no_grad()
    def sample_heun(
        self,
        clean_mel_ori: torch.Tensor,
        T_tar: int,
        audio_tokens: torch.Tensor,
        spk_emb: torch.Tensor,
        n_steps: int = 10,
    ) -> torch.Tensor:
        """
        Heun 方法（改进的欧拉法），二阶精度
        x_{t+Δt} = x_t + 0.5 * Δt * (v_t + v_{t+Δt})
        
        比 Euler 法更精确，但需要两次模型前向传播
        """
        B = clean_mel_ori.shape[0]
        n_mels = clean_mel_ori.shape[1]
        device = clean_mel_ori.device
        dtype = clean_mel_ori.dtype

        mel_t = torch.randn(B, n_mels, T_tar, device=device, dtype=dtype)
        dt = 1.0 / n_steps
        time_steps = torch.linspace(0, 1, n_steps + 1, device=device, dtype=dtype)

        for i in range(n_steps):
            t = time_steps[i]
            t_next = time_steps[i + 1]
            t_batch = t.expand(B)
            t_next_batch = t_next.expand(B)

            # k1 = v(x_t, t)
            k1, _ = self.model(
                clean_mel_ori=clean_mel_ori,
                noisy_mel_tar=mel_t,
                audio_tokens=audio_tokens,
                t=t_batch,
                spk_emb=spk_emb,
                masked_mel_tar=None,
                noisy_mel_ori=clean_mel_ori,
            )

            # x_trial = x_t + dt * k1
            mel_trial = mel_t + dt * k1

            # k2 = v(x_trial, t+dt)
            k2, _ = self.model(
                clean_mel_ori=clean_mel_ori,
                noisy_mel_tar=mel_trial,
                audio_tokens=audio_tokens,
                t=t_next_batch,
                spk_emb=spk_emb,
                masked_mel_tar=None,
                noisy_mel_ori=clean_mel_ori,
            )

            # Heun 更新: x_{t+dt} = x_t + 0.5 * dt * (k1 + k2)
            mel_t = mel_t + 0.5 * dt * (k1 + k2)

        return mel_t
