# 扩散模型

## 什么是扩散？

逐步加噪，逐步去噪。

## 数学原理

**DDPM（Denoising Diffusion Probabilistic Models）** ：

- 前向加噪过程：

给定数据 $x_0 \sim q(x_0)$，前向过程通过马尔可夫链逐步加噪：
$$
q(x_t|x_{t-1}) = \mathcal{N}\big(x_t; \sqrt{1-\beta_t}\,x_{t-1},\,\beta_t \mathbf{I}\big)
$$
其中 $\beta_1 < \beta_2 < \dots < \beta_T$ 是预先设定的噪声调度（noise schedule）。

>   重参数化技巧，可以直接一步计算任意时刻 $t$ 的带噪样本（无需迭代）：
>   $$x_t = \sqrt{\bar{\alpha}_t}\,x_0 + \sqrt{1-\bar{\alpha}_t}\,\epsilon,\quad \epsilon \sim \mathcal{N}(0,\mathbf{I})$$
>   其中 $\alpha_t = 1-\beta_t,\ \bar{\alpha}_t = \prod_{s=1}^{t}\alpha_s$。


- 反向去噪过程（训练目标）：

模型学习预测噪声 $\epsilon_\theta(x_t,t)$，训练目标为最小化预测噪声与真实噪声的 MSE：

$$
L_{\text{simple}} = \mathbb{E}_{t,x_0,\epsilon}\Big[\big\|\epsilon - \epsilon_\theta(x_t,t)\big\|^2\Big]
$$

>   速度预测（v‑prediction）变体\
>   更稳定的训练目标。速度 $v_t$ 定义为：
>   $$v_t = \sqrt{\bar{\alpha}_t}\,\epsilon - \sqrt{1-\bar{\alpha}_t}\,x_0$$
>   对应的 $x_0$ 重建公式为：
>   $$x_0 = \sqrt{\bar{\alpha}_t}\,x_t - \sqrt{1-\bar{\alpha}_t}\,v_\theta(x_t,t)$$


### DiT

- **单流** ：

输入：`[条件token | 音频潜在token]`

- **双流** ：

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import math
from typing import Optional, Tuple

class DiffusionMixin:
    """
    DDPM 扩散过程的 Mixin 类
    包含：噪声调度、加噪过程、目标计算、损失计算、采样
    """

    def __init__(
        self,
        num_timesteps: int = 1000,
        noise_schedule: str = "scaled_linear",
        beta_start: float = 0.00085,
        beta_end: float = 0.012,
        prediction_type: str = "v_prediction",  # v-prediction 更稳定
        snr_gamma: float = 5.0,  # Min-SNR 加权
        **kwargs,
    ):
        super().__init__(**kwargs)
        self.num_timesteps = num_timesteps
        self.prediction_type = prediction_type
        self.snr_gamma = snr_gamma

        # ===== 1. 构建噪声调度 (beta schedule) =====
        if noise_schedule == "scaled_linear":
            betas = torch.linspace(
                beta_start**0.5, beta_end**0.5, num_timesteps, dtype=torch.float32
            ) ** 2
        elif noise_schedule == "linear":
            betas = torch.linspace(beta_start, beta_end, num_timesteps, dtype=torch.float32)
        else:
            raise ValueError(f"Unknown noise schedule: {noise_schedule}")

        alphas = 1.0 - betas
        alphas_cumprod = torch.cumprod(alphas, dim=0)  # \bar{\alpha}_t

        # 注册为 buffer（不训练，但随模型保存）
        self.register_buffer("betas", betas)
        self.register_buffer("alphas", alphas)
        self.register_buffer("alphas_cumprod", alphas_cumprod)

        # 预计算速度预测相关量
        self.register_buffer("sqrt_alphas_cumprod", torch.sqrt(alphas_cumprod))
        self.register_buffer(
            "sqrt_one_minus_alphas_cumprod", torch.sqrt(1.0 - alphas_cumprod)
        )

    # ===== 2. 前向加噪：一步计算 x_t =====
    def add_noise(
        self,
        original_samples: torch.Tensor,
        noise: torch.Tensor,
        timesteps: torch.Tensor,
    ) -> torch.Tensor:
        """
        前向扩散：x_t = sqrt(ᾱ_t) * x_0 + sqrt(1 - ᾱ_t) * ε
        
        Args:
            original_samples: x_0, [B, C, T]
            noise: ε ~ N(0, I), 同 shape
            timesteps: t, [B]
        Returns:
            noisy_samples: x_t, [B, C, T]
        """
        sqrt_alpha_prod = self.sqrt_alphas_cumprod[timesteps]
        sqrt_one_minus_alpha_prod = self.sqrt_one_minus_alphas_cumprod[timesteps]

        # 扩展维度以匹配 [B, C, T]
        while len(sqrt_alpha_prod.shape) < len(original_samples.shape):
            sqrt_alpha_prod = sqrt_alpha_prod.unsqueeze(-1)
            sqrt_one_minus_alpha_prod = sqrt_one_minus_alpha_prod.unsqueeze(-1)

        noisy_samples = (
            sqrt_alpha_prod * original_samples + sqrt_one_minus_alpha_prod * noise
        )
        return noisy_samples

    # ===== 3. 计算训练目标（速度预测） =====
    def get_velocity(
        self, sample: torch.Tensor, noise: torch.Tensor, timesteps: torch.Tensor
    ) -> torch.Tensor:
        """
        速度目标: v_t = sqrt(ᾱ_t) * ε - sqrt(1 - ᾱ_t) * x_0
        """
        sqrt_alpha_prod = self.sqrt_alphas_cumprod[timesteps]
        sqrt_one_minus_alpha_prod = self.sqrt_one_minus_alphas_cumprod[timesteps]

        while len(sqrt_alpha_prod.shape) < len(sample.shape):
            sqrt_alpha_prod = sqrt_alpha_prod.unsqueeze(-1)
            sqrt_one_minus_alpha_prod = sqrt_one_minus_alpha_prod.unsqueeze(-1)

        velocity = sqrt_alpha_prod * noise - sqrt_one_minus_alpha_prod * sample
        return velocity

    # ===== 4. 从模型预测中恢复 x_0 =====
    def predict_original_from_velocity(
        self,
        sample: torch.Tensor,
        model_output: torch.Tensor,
        timesteps: torch.Tensor,
    ) -> torch.Tensor:
        """
        x_0 = sqrt(ᾱ_t) * x_t - sqrt(1 - ᾱ_t) * v_θ(x_t, t)
        """
        sqrt_alpha_prod = self.sqrt_alphas_cumprod[timesteps]
        sqrt_one_minus_alpha_prod = self.sqrt_one_minus_alphas_cumprod[timesteps]

        while len(sqrt_alpha_prod.shape) < len(sample.shape):
            sqrt_alpha_prod = sqrt_alpha_prod.unsqueeze(-1)
            sqrt_one_minus_alpha_prod = sqrt_one_minus_alpha_prod.unsqueeze(-1)

        pred_original_sample = (
            sqrt_alpha_prod * sample - sqrt_one_minus_alpha_prod * model_output
        )
        return pred_original_sample

    # ===== 5. Min-SNR 加权损失 =====
    def compute_loss(
        self,
        model_output: torch.Tensor,
        target: torch.Tensor,
        timesteps: torch.Tensor,
    ) -> torch.Tensor:
        """
        Min-SNR 加权 MSE 损失
        
        SNR(t) = ᾱ_t / (1 - ᾱ_t)
        weight = min(SNR(t), γ) / SNR(t)
        loss = weight * MSE(v_θ, v_target)
        """
        mse_loss = F.mse_loss(model_output.float(), target.float(), reduction="none")
        mse_loss = mse_loss.mean(dim=list(range(1, len(mse_loss.shape))))  # [B]

        # 计算 SNR
        alphas_cumprod = self.alphas_cumprod[timesteps]
        snr = alphas_cumprod / (1 - alphas_cumprod)

        # Min-SNR 截断
        maybe_clipped_snr = snr.clone()
        maybe_clipped_snr = torch.min(
            maybe_clipped_snr, torch.full_like(maybe_clipped_snr, self.snr_gamma)
        )

        if self.prediction_type == "v_prediction":
            # v-prediction: weight = min(SNR, γ) / SNR  等价于 min(SNR, γ) * (1-ᾱ) / ᾱ ... 简化形式
            # 标准形式: weight = min(SNR, gamma) / SNR
            loss_weights = maybe_clipped_snr / snr
        else:
            loss_weights = torch.ones_like(snr)

        loss = (loss_weights * mse_loss).mean()
        return loss

    # ===== 6. DDIM 采样（少步推理） =====
    @torch.no_grad()
    def sample(
        self,
        model,
        latents_shape: Tuple[int, ...],
        condition: dict,
        num_inference_steps: int = 50,
        guidance_scale: float = 5.0,
        device=None,
    ) -> torch.Tensor:
        """
        DDIM 采样：从 x_T ~ N(0,I) 开始，逐步去噪到 x_0
        
        使用 DDIM 确定性采样器，比 DDPM 的 T=1000 步快得多
        """
        if device is None:
            device = self.betas.device

        # 初始化: x_T ~ N(0, I)
        latents = torch.randn(latents_shape, device=device, dtype=torch.float32)

        # 构建时间步序列（均匀子采样）
        step_ratio = self.num_timesteps // num_inference_steps
        timesteps = list(
            range(0, self.num_timesteps, step_ratio)
        )
        timesteps = torch.tensor(timesteps, device=device).long()

        # 从 T 到 0 逆向采样
        for i in reversed(range(len(timesteps))):
            t = timesteps[i]
            t_prev = timesteps[i - 1] if i > 0 else torch.tensor(-1, device=device)

            # ---- 无分类器引导 (CFG) ----
            # 两次前向：有条件 + 无条件
            if guidance_scale != 1.0:
                # 条件预测
                noise_pred_cond = model(latents, t, **condition)
                # 无条件预测（条件置空）
                uncond_condition = {k: self._make_uncond(v) for k, v in condition.items()}
                noise_pred_uncond = model(latents, t, **uncond_condition)
                # 引导插值
                noise_pred = noise_pred_uncond + guidance_scale * (
                    noise_pred_cond - noise_pred_uncond
                )
            else:
                noise_pred = model(latents, t, **condition)

            # ---- DDIM 一步更新 ----
            alpha_prod_t = self.alphas_cumprod[t]
            alpha_prod_t_prev = (
                self.alphas_cumprod[t_prev]
                if t_prev >= 0
                else torch.tensor(1.0, device=device)
            )

            # 从速度预测重建 x_0
            pred_original_sample = self.predict_original_from_velocity(
                latents, noise_pred, t.unsqueeze(0)
            )

            # 计算 x_{t-1} 的方向
            beta_prod_t = 1 - alpha_prod_t
            beta_prod_t_prev = 1 - alpha_prod_t_prev

            pred_sample_direction = torch.sqrt(beta_prod_t_prev) * (
                (latents - torch.sqrt(alpha_prod_t) * pred_original_sample)
                / torch.sqrt(beta_prod_t)
            )

            # DDIM 更新（确定性，无随机性）
            latents = (
                torch.sqrt(alpha_prod_t_prev) * pred_original_sample
                + pred_sample_direction
            )

        return latents
```
